```python
SCRIPT_DISJOINT_CONFIRMED = {
    "Cantonese", "Chinese", "Japanese", "Korean", "Russian", "Arabic"
}
NEEDS_VERIFICATION = {"Hindi"}   # confirm script before routing to unicode
NEEDS_MODEL = {
    "German", "Philippines", "Vietnamese", "Malay", "Spanish", "French", "Italian"
}

def route(first_language: str) -> str:
    if first_language in SCRIPT_DISJOINT_CONFIRMED:
        return "unicode"
    if first_language in NEEDS_VERIFICATION:
        return "unicode_pending_check"
    return "model"
```

## Tokenizer (`src/tokenizer.py`)

CJK doesn't tokenize on whitespace — a run of Han/Kana/Hangul characters
between two English words may contain several words with no space. Tokenize
CJK spans at the character level, Latin spans at the word level, in one pass:

```python
import regex as re

def tokenize_mixed(text: str) -> list[str]:
    """Split text into CJK-character tokens and Latin-word tokens."""
    pattern = (
        r'\p{Script=Han}|\p{Script=Hiragana}|\p{Script=Katakana}'
        r'|\p{Script=Hangul}|[A-Za-z]+(?:\'[A-Za-z]+)?|[^\sA-Za-z]'
    )
    return re.findall(pattern, text)
```

## Unicode tagger (`src/unicode_tagger.py`)

```python
import regex as re

_SCRIPTS = ["Han", "Hiragana", "Katakana", "Hangul", "Devanagari", "Cyrillic", "Arabic", "Latin"]

def char_script(ch: str) -> str:
    for name in _SCRIPTS:
        if re.match(fr'\p{{Script={name}}}', ch):
            return name
    return "other"

def tag_token_unicode(token: str) -> str:
    """Return the dominant script name for a token, or 'other' if mixed/unscripted."""
    scripts = {char_script(c) for c in token if not c.isspace()}
    scripts.discard("other")
    if len(scripts) == 1:
        return scripts.pop()
    return "other"
```
Map the returned script name back to the row's actual `first_language` /
`second_language` value (e.g. `Han` -> row's `first_language`, `Latin` ->
row's `second_language`).

## Model tagger (`src/model_tagger.py`)

**V1 — constrained fastText, no training required:**
```python
import fasttext
lid = fasttext.load_model("lid.176.bin")  # download from fasttext.cc

LANG_TO_CODE = {"German":"de","Philippines":"tl","Vietnamese":"vi","Malay":"ms",
                 "Spanish":"es","French":"fr","Italian":"it","English":"en"}

def tag_token_constrained(token: str, lang_a: str, lang_b: str) -> str:
    """Constrain fastText's 176-way prediction to the two known candidates."""
    labels, probs = lid.predict(token, k=176)
    scores = {l.replace("__label__", ""): p for l, p in zip(labels, probs)}
    code_a, code_b = LANG_TO_CODE[lang_a], LANG_TO_CODE[lang_b]
    return lang_a if scores.get(code_a, 0) >= scores.get(code_b, 0) else lang_b
```

**Escalation rule (V1 -> V2) — explicit, not a vibe:**
Manually label ~100-150 tokens per language pair. If per-pair token accuracy
is **below 90%**, escalate that pair to V2. Expect German/Malay/Tagalog to
clear 90% easily; expect Spanish/French/Italian-English to be the ones that
don't, because of shared Romance function words ("no", "la", "en", "un").

**V2 — weak-supervised XLM-R, only for pairs that fail the 90% bar:**
- Use `wordfreq` per-language wordlists to auto-label unambiguous tokens
  (common in language A, absent from language B -> weak label A).
- Fine-tune `xlm-roberta-base` as a token classifier on the weak-labeled
  spans; prepend a marker token for the row's known language pair.
- For Spanish-English specifically, use the LinCE benchmark's real labeled
  data instead of weak supervision — no need to bootstrap what already exists.

## Segments (`src/segments.py`)

```python
def to_segments(tagged_tokens: list[tuple[str, str]]) -> list[list]:
    """Collapse consecutive same-language tokens into [language, length] runs."""
    segments = []
    for _, lang in tagged_tokens:
        if segments and segments[-1][0] == lang:
            segments[-1][1] += 1
        else:
            segments.append([lang, 1])
    return segments
```
Decide up front whether an `"other"` tag (numbers, punctuation) breaks a run
or gets absorbed into the surrounding language's run — absorbing is usually
cleaner for the downstream goals below.

## Tests (`tests/test_segments.py`)

```python
from src.segments import to_segments

def test_segments_reconstruct_token_count():
    tagged = [("a","en"),("b","en"),("c","hi"),("d","hi"),("e","hi"),("f","en")]
    segs = to_segments(tagged)
    assert sum(length for _, length in segs) == len(tagged)
    assert segs == [["en", 2], ["hi", 3], ["en", 1]]

def test_no_adjacent_duplicate_languages():
    tagged = [("a","en"),("b","hi"),("c","en")]
    segs = to_segments(tagged)
    langs = [lang for lang, _ in segs]
    assert all(langs[i] != langs[i+1] for i in range(len(langs)-1))
```
